Scheduler

interface SchedulerInterface
{
    /** @throws InvalidArgumentException nevalidní cron výraz nebo duplicitní jméno */
    public function add(ScheduledJob $job): void;

    /** Spustí joby, které jsou na řadě v aktuální minutě. Volá systémový cron každou minutu. */
    public function runDue(): void;
}

interface JobInterface
{
    /** Výjimka = běh selhal; ostatní joby na řadě poběží dál. */
    public function run(): void;
}

ScheduledJob

final readonly class ScheduledJob
{
    public function __construct(
        public string $name,
        public string $cron,
        public JobInterface $job,
        public bool $withoutOverlapping = true,
        public int $lockTtlSeconds = 3600,
        public ?DateTimeZone $timezone = null,
    ) {}

    public function withJob(JobInterface $job): self;
}

Mutex

/**
 * Každá implementace musí zaručit:
 *  - acquire je atomický napříč procesy a servery se sdíleným backendem
 *  - každý zámek po TTL vyprší (žádné mrtvé zámky po kill -9 / OOM / deployi)
 *  - release smaže jen vlastní zámek (kontrola tokenu)
 *  - výpadek backendu = MutexException, nikdy „asi mám zámek“
 */
interface MutexInterface
{
    /** @return Lock|null  null = zámek drží někdo jiný
     *  @throws MutexException */
    public function acquire(string $key, int $ttlSeconds): ?Lock;

    /** Idempotentní; vypršelý nebo cizí zámek nechá být.
     *  @throws MutexException */
    public function release(Lock $lock): void;
}

final readonly class Lock
{
    public function __construct(public string $key, public string $token) {}
}

final class MutexException extends RuntimeException {}

Použití

$factory = new SchedulerFactory(new RedisMutex($redis), $logger, $clock);   // PSR-3 logger, PSR-20 clock
$scheduler = $factory->create($config['scheduler_driver']);                 // 'laravel'

$scheduler->add(new ScheduledJob('billing.sync-invoices', '*/15 * * * *', $syncInvoicesJob, lockTtlSeconds: 900));
$scheduler->add(new ScheduledJob(
    'billing.monthly-report',
    '0 6 1 * *',
    $monthlyReportJob,
    timezone: new DateTimeZone('Europe/Prague'),
));

$scheduler->runDue();

Crontab: * * * * * php /app/bin/scheduler-run.
